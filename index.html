function login() { const u = document.getElementById('uInp').value.trim(); const p = document.getElementById('pInp').value.trim(); // تنبيه بسيط للتأكد أن الزر استجاب للضغط if(!u || !p) { alert("يرجى إدخال الاسم وكلمة المرور"); return; } console.log("جاري محاولة الدخول..."); // للتأكد في خلفية المتصفح db.ref('users/'+u).once('value').then(s => { const d = s.val(); if(d && d.pass !== p) { alert("كلمة المرور خاطئة!"); } else { user = d || { name: u, pass: p, tag: u+"#"+Math.floor(1000+Math.random()*9000), friends: {} }; if(!d) db.ref('users/'+u).set(user); localStorage.setItem('app_user', JSON.stringify(user)); // التوجيه المباشر بدل التحديث لضمان السرعة document.getElementById('authArea').classList.add('hidden'); document.getElementById('appArea').classList.remove('hidden'); location.reload(); // تحديث نهائي لتنشيط البيانات } }).catch(err => { alert("خطأ في الاتصال بقاعدة البيانات: " + err.message); }); }
